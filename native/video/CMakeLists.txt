cmake_minimum_required(VERSION 3.16)
project(mst2_nar_video C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(dcc_ts STATIC src/dcc_ts.c src/dcc_video_slot.c src/dcc_avc.c src/dcc_arm32_hook.c src/dcc_screen.c src/dcc_video_lifecycle.c src/dcc_screen_session.c)
target_include_directories(dcc_ts PUBLIC include)
if(MSVC)
  target_compile_options(dcc_ts PRIVATE /W4 /WX)
else()
  target_compile_options(dcc_ts PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_executable(ts_test tests/ts_test.c)
target_link_libraries(ts_test PRIVATE dcc_ts)
add_test(NAME ts_test COMMAND ts_test)
add_executable(slot_test tests/slot_test.c)
target_link_libraries(slot_test PRIVATE dcc_ts)
add_test(NAME slot_test COMMAND slot_test)
add_executable(avc_test tests/avc_test.c)
target_link_libraries(avc_test PRIVATE dcc_ts)
add_test(NAME avc_test COMMAND avc_test)
add_executable(arm32_hook_test tests/arm32_hook_test.c)
target_link_libraries(arm32_hook_test PRIVATE dcc_ts)
add_test(NAME arm32_hook_test COMMAND arm32_hook_test)
add_executable(screen_test tests/screen_test.c)
target_link_libraries(screen_test PRIVATE dcc_ts)
add_test(NAME screen_test COMMAND screen_test)
add_executable(lifecycle_test tests/lifecycle_test.c)
target_link_libraries(lifecycle_test PRIVATE dcc_ts)
add_test(NAME lifecycle_test COMMAND lifecycle_test)
add_executable(screen_session_test tests/screen_session_test.c)
target_link_libraries(screen_session_test PRIVATE dcc_ts)
add_test(NAME screen_session_test COMMAND screen_session_test)
